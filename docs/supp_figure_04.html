# Read+combine one ambient-tool subfolder's doublet calls with the dataset's ground-truth metadata.
load_ambient_results <- function(path, uncorrected_dir, filter_remove = FALSE) {

  # ground-truth metadata always comes from the uncorrected/base folder for this dataset,
  # regardless of which ambient-tool subfolder is currently being processed
  meta <- read_tsv(file.path(dirname(path), uncorrected_dir, "metadata.tsv"), show_col_types = FALSE)
  meta$doublet_class[meta$doublet_class == "multiplet"] <- "doublet"
  
  #filter metadata based on uncorrected qc filtered barcodes - these are droplets that pass QC (based on RNA reads) AND have a doublet/singlet assignment based on demultiplexing
  barcodes_to_keep <- read_tsv(file.path(dirname(path), uncorrected_dir, "filtered_qc_barcodes.tsv"), show_col_types = F) %>% pull(.)
  
  meta <- meta %>% 
    filter(., Barcode %in% barcodes_to_keep)

  # doublet calling results for this ambient-tool setting
  doublet_file <- list.files(path = file.path(path, "doublet_calls"),
                             pattern = "consensus", full.names = TRUE)
  dbl_res <- read_csv(doublet_file, show_col_types = FALSE)
  combined_res <- right_join(dbl_res, meta)

  #replace missing doublet (NAs) labels with "removed_cell"- using scDblFinder to guide but will be the same with all tools
  combined_res$cells_kept <- ifelse(is.na(combined_res$scDblFinder_DropletType), "removed_cell", "kept_cell")

  #if this setting is ambient-corrected, bring in its ncount/nfeature "removed" columns too,
  # prefixed with the tool name so they sit side by side once every setting is combined
  meta_current <- read_tsv(file.path(path, "metadata.tsv"), show_col_types = FALSE)
  if ("ncount_frac_removed" %in% colnames(meta_current)) {
    ambient_tool_name <- gsub(paste0(uncorrected_dir, "_"), "", basename(path), fixed = TRUE)
    meta_current <- meta_current %>%
      select(Barcode, ncount, nfeatures_rounded, nfeature_frac_removed, ncount_frac_removed)
    colnames(meta_current)[2:5] <- paste0(ambient_tool_name, "_", colnames(meta_current)[2:5])
    combined_res <- left_join(combined_res, meta_current)
  }

  combined_res$ambient_tool <- if (basename(path) == uncorrected_dir) {
    "uncorrected"
  } else {
    gsub(paste0(uncorrected_dir, "_"), "", basename(path), fixed = TRUE)
  }

  return(combined_res)
}


load_dataset_results <- function(cfg) {
  files <- list.dirs(path = here("output/data", cfg$data_dir), full.names = TRUE, recursive = FALSE)
  files <- files[grepl(cfg$sample_grep, files)]

  list_results <- lapply(files, load_ambient_results,
                         uncorrected_dir = cfg$uncorrected_dir)
  names(list_results) <- basename(files)

  df <- bind_rows(list_results)
  df$dataset <- cfg$dataset_label
  return(df)
}

# Load a dataset's precomputed doublet-calling performance metrics (see *_evaluate_doublet_calling.Rmd).
load_performance_results <- function(cfg) {
  perf <- read_csv(here("output/results", cfg$performance_file),
                   show_col_types = FALSE, col_types = c("ccnnnnnnnnnnnn"))
  # perf[is.na(perf)] <- 0

  perf <- perf %>%
    mutate(doublet_tool = factor(doublet_tool,
                                 levels = c("scDblFinder", "scds", "cxds", "bcds", "scrublet", "DoubletDetection")),
           ambient_tool = factor(ambient_tool,
                                 levels = rev(c("uncorrected", "CellBender", "soupX", "decontX", "decontX_BG"))))
  perf$dataset <- cfg$dataset_label
  return(perf)
}

run_assignment_evaluation_heatmap <- function(results_df,
                                      plotting_metric,
                                      name_metric,
                                      name_scale,
                                      dataset_label,
                                      limits = NULL) {
  results_df <- results_df %>%
    dplyr::rename(plotting_metric = plotting_metric)

  heatmap <- results_df %>%
    ggplot() +
    geom_tile(aes(x = doublet_tool, y = ambient_tool, fill = plotting_metric), color = "black", size = 0.1) +
    geom_text(aes(x = doublet_tool, y = ambient_tool, label = round(plotting_metric, 2)), size = 5) +
    theme_bw() + plot_sizing_keep_legend + rotate_x + title_size +
    ylab("ambient setting") + xlab("doublet tool") +
    scale_fill_distiller(palette = "RdBu", direction = -1, name = name_scale, limits = limits) +
    ggtitle(dataset_label, subtitle = name_metric) + title_size

  return(heatmap)
}

analyse_mislabelled_doublets <- function(df_plot,
                                         ambient_tools = setdiff(unique(df_plot$ambient_tool), "uncorrected"),
                                         doublet_tool,
                                         dataset_label) {
  doublet_tool_column <- paste0(doublet_tool, "_DropletType")
  list_unidentified_doublets <- lapply(split(df_plot, df_plot$ambient_tool), function(entry) {
    entry %>% dplyr::rename(doublet_tool_column = doublet_tool_column) %>%
      filter(doublet_class == "doublet", doublet_tool_column != "doublet") %>% pull(Barcode)
  })

  m_dist <- make_comb_mat(list_unidentified_doublets)
  full_comb <- paste(rep("1", length(list_unidentified_doublets)), collapse = "")
  missed_doublets <- extract_comb(m_dist, full_comb)

  list_missed_doublets_meta <- lapply(ambient_tools, function(TOOL) {
    only_here <- setdiff(list_unidentified_doublets[[TOOL]], missed_doublets)
    df_plot %>% filter(Barcode %in% only_here) %>% mutate(status = TOOL)
  })

  overview <- df_plot %>%
    mutate(status = case_when(Barcode %in% missed_doublets ~ "never detected",
                              TRUE ~ "remove")) %>%
    filter(status != "remove")

  meta_total <- bind_rows(list_missed_doublets_meta, overview)

  # ---- stacked barplot ----
  missed_all_n <- sum(meta_total$status[meta_total$ambient_tool == "uncorrected"] == "never detected")
  missed_all_label <- paste0("never detected\nn=", missed_all_n)

  ambient_tools_colours_nocounts <- c("grey80", "tan")
  names(ambient_tools_colours_nocounts) <- c(missed_all_label, "in setting")

  special <- meta_total %>% filter(., ambient_tool == "uncorrected") %>% dplyr::count(status) %>% filter(status == "never detected")
  main    <- meta_total %>% filter(., ambient_tool == "uncorrected") %>% dplyr::count(status) %>% filter(status %in% ambient_tools)

  result <- map_dfr(main$status, function(s) {
    bind_rows(main %>% filter(status == s), special) %>% mutate(tool = s, .before = 1)
  }) %>%
    mutate(status = factor(case_when(status == "never detected" ~ missed_all_label, TRUE ~ "in setting"),
                           levels = rev(c(missed_all_label, "in setting"))),
           label = ifelse(status == "in setting", n, NA))

  stacked_barplot <- result %>%
    ggplot(aes(x = tool, y = n, fill = status)) +
    geom_bar(position = "stack", stat = "identity") +
    geom_text(position = position_stack(vjust = 0.5), aes(label = label), size = 5) +
    plot_sizing + rotate_x +
    scale_fill_manual(values = ambient_tools_colours_nocounts) +
    ylab("# doublets") +
    theme(legend.position = "bottom") +
    ggtitle(dataset_label, subtitle = paste0("Undetected doublets\n", doublet_tool)) + title_size

  # ---- per-tool boxplots: single facet_wrap with free y-axes and shared title ----

  prop_long <- map_dfr(ambient_tools, function(TOOL) {
    prop_removed_name <- paste0(TOOL, "_ncount_frac_removed")
    meta_total %>%
      filter(status %in% c(TOOL, "never detected")) %>%
      dplyr::rename(prop_removed = all_of(prop_removed_name)) %>%
      mutate(facet_label = factor(paste0("corrected using\n", TOOL),
                                  levels = paste0("corrected using\n", ambient_tools))) %>%
      select(status, prop_removed, facet_label)
  }) %>%
    # "never detected" first, then tools in ambient_tools order — applied after
    # bind_rows so all rows share consistent factor levels
    mutate(status = factor(status, levels = c("never detected", ambient_tools)))

  per_tool_combined <- prop_long %>%
    ggplot(aes(x = status, y = prop_removed, fill = status)) +
    geom_boxplot( outliers = FALSE) +
    geom_jitter(aes(x = status, y = prop_removed), size = 0.3, alpha = 0.2, width = 0.2) +
    facet_wrap(~facet_label, scales = "free", nrow = 1) +
    plot_sizing + strip_fix + rotate_x +
    scale_fill_manual(values = ambient_tools_colours) +
    xlab("") + ylab("proportion removed") + theme(legend.position = "none",
                                                  strip.text = element_text(size = 18))

  per_tool_plot <- per_tool_combined +
    plot_annotation(title = dataset_label,
                    subtitle = paste0("Ambient correction in undetected doublets\n", doublet_tool),
                    theme = title_size)

  list(stacked_barplot    = stacked_barplot,
       per_tool_plot      = per_tool_plot,
       per_tool_plot_list = list(per_tool_combined))
}

generate_dataset_figures <- function(cfg) {
  df_plot            <- load_dataset_results(cfg)
  performance_results <- load_performance_results(cfg)

  mislabeled_doublets_scdblfinder <- analyse_mislabelled_doublets(
    df_plot       = df_plot,
    doublet_tool  = "scDblFinder",
    dataset_label = cfg$dataset_label
  )

  list(
    dataset_label          = cfg$dataset_label,
    performance_results    = performance_results,
    performance_plot_mcc   = run_assignment_evaluation_heatmap(
      results_df     = performance_results,
      plotting_metric = "mcc",
      name_metric    = "MCC",
      name_scale     = "MCC",
      dataset_label  = cfg$dataset_label
    ),
    undetected_stacked       = mislabeled_doublets_scdblfinder$stacked_barplot,
    undetected_per_tool      = mislabeled_doublets_scdblfinder$per_tool_plot,
    undetected_per_tool_list = mislabeled_doublets_scdblfinder$per_tool_plot_list
  )
}
